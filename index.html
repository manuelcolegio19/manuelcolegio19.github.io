        updateCart();

        openCart();

      });

    /* =======================================================
       UPDATE CART
    ======================================================== */

    function updateCart() {

      cartCount.textContent = cart.length;

      if (cart.length === 0) {

        cartItems.innerHTML = `
          <div class="empty-cart">
            <div>
              <div style="font-size:35px;margin-bottom:12px;">🛒</div>
              <p>Tu carrito está vacío.</p>
            </div>
          </div>
        `;

        cartTotal.textContent = "0,00€";

        element.className = "cart-item";

        element.innerHTML = `
          <img src="${item.image}" alt="${item.name}">

          <div>
            <div class="cart-item-name">
              ${item.name}
            </div>

            <div class="cart-item-price">

            data-index="${index}"

          >
            ✕
          </button>
        `;

        cartItems.appendChild(element);

      });

      cartTotal.textContent =
        total.toFixed(2).replace(".", ",") + "€";


      document.querySelectorAll(".remove-item").forEach(button => {

          const index = Number(button.dataset.index);

       SEARCH LIVE
    ======================================================== */

    searchInput.addEventListener("input", () => {

      const query =
        searchInput.value.toLowerCase().trim();

      if (!query) {

        searchResults.textContent =
          "Empieza a escribir para buscar.";

        return;
      }

      let found = 0;

      products.forEach(product => {

        const name =
          product
            .querySelector(".product-name")
            .textContent
            .toLowerCase();

        const category =
          product
            .querySelector(".product-category")
            .textContent
            .toLowerCase();

        if (
          name.includes(query) ||
          category.includes(query)
        ) {

          found++;

        }

      });

      if (found === 0) {

        searchResults.innerHTML =
          `No encontramos resultados para <strong>"${query}"</strong>.`;

      } else {

        searchResults.innerHTML =
          `Encontramos <strong>${found}</strong> producto${found > 1 ? "s" : ""}.`;

      }

    });

🚀 Para subirlo a GitHub

  1. Crea un repositorio, por ejemplo solehype.

  2. Crea un archivo llamado index.html.

  3. Pega todo el código de arriba.

  4. Haz Commit changes.

  5. Ve a Settings → Pages.

  6. En Source, selecciona Deploy from a branch.

  7. Selecciona main y /root.

  8. Guarda.

Y tendrás tu tienda funcionando en GitHub Pages.

Importante: las imágenes del ejemplo vienen de URLs externas para que el HTML funcione sin una carpeta images/. Para una web comercial real conviene sustituirlas por imágenes propias/licenciadas y conectar el checkout a un backend/pasarela de pago.
